def is_perfect_square(x: int) -> bool:
    if x < 0:
        return False
    if x == 0:
        return True
    left, right = 1, x
    while left <= right:
        mid = (left + right) // 2
        if mid * mid == x:
            return True
        elif mid * mid < x:
            left = mid + 1
        else:
            right = mid - 1
    return False

# Example usage:
print(is_perfect_square(16))  # True
print(is_perfect_square(14))  # False
print(is_perfect_square(0))   # True
print(is_perfect_square(1))   # True
