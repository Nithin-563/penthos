import re

def rle(data: str) -> str:
    return ''.join(f'{char}{len(list(group))}' if len(list(group)) > 1 else char for char, group in re.findall(r'(.)\1*', data))

def rle_decode(s: str) -> str:
    return ''.join(char * int(count) if count else char for char, count in re.findall(r'([^\d])(\d*)', s))
