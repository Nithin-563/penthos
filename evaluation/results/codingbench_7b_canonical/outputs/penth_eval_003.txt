def is_balanced(s: str) -> bool:
    stack = []
    mapping = {")": "(", "]": "[", "}": "{"}
    
    for char in s:
        if char in mapping.values():
            stack.append(char)
        elif char in mapping:
            if not stack or mapping[char] != stack.pop():
                return False
    
    return not stack

# Example test cases
print(is_balanced("(x + y)"))  # True
print(is_balanced("([{}])"))  # True
print(is_balanced("([)]"))    # False
print(is_balanced(")"))       # False
