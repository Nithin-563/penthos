from typing import List

def missing_and_dup(nums: List[int]) -> List[int]:
    n = len(nums)
    total_sum = n * (n + 1) // 2
    total_dup = sum(nums) - total_sum
    xor_sum = 0
    for num in nums:
        xor_sum ^= num
    for i in range(1, n + 1):
        xor_sum ^= i
    diff = xor_sum & -xor_sum
    missing = 0
    for num in nums:
        if num & diff:
            missing ^= num
    for i in range(1, n + 1):
        if i & diff:
            missing ^= i
    return [missing, missing + total_dup]

# Example usage:
# print(missing_and_dup([1,2,2,4]))  # Output: [3, 2]
# print(missing_and_dup([3,1,3]))    # Output: [2, 3]
