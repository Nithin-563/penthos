from typing import List

def near_duplicate(nums: List[int], k: int) -> bool:
    num_indices = {}
    for i, num in enumerate(nums):
        if num in num_indices:
            if i - num_indices[num] <= k:
                return True
        num_indices[num] = i
    return False

# Example usage:
# print(near_duplicate([1,2,3,1], 3))  # Output: True
# print(near_duplicate([1,0,1,1], 1))  # Output: True
# print(near_duplicate([1,2,3,1,2,3], 2))  # Output: False
